# Harm event code often calls this subroutine, dozens or hundreds of time per
#  year per character
ek_human_age_equivalent = {
	if = {
		limit = { age > 18 }
		if = {
			limit = { has_variable = ek_human_age_equivalent_cache }
			value = var:ek_human_age_equivalent_cache
		}
		else = {
			# Hopefully should not come here too often
			value = ek_human_age_equivalent_calc
			min = 18
		}
	}
	else = {
		value = age
	}
}
